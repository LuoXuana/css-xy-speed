cmake_minimum_required(VERSION 3.24)
project(CSSXYSpeed VERSION 0.1.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CSS XY Speed requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
set(IMGUI third_party/imgui)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
  third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
set(HUD_RUNTIME src/engine_bridge.cpp ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp
  ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_dx9.cpp)
add_library(CSSXYSpeed SHARED src/xy_speed.cpp ${HUD_RUNTIME})
target_include_directories(CSSXYSpeed PRIVATE include ${IMGUI})
target_link_libraries(CSSXYSpeed PRIVATE minhook d3d9 user32)
add_executable(xy_speed_loader src/loader.cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(xy_speed_window_tests tests/xy_speed_window_tests.cpp ${HUD_RUNTIME})
  target_include_directories(xy_speed_window_tests PRIVATE include ${IMGUI})
  target_link_libraries(xy_speed_window_tests PRIVATE minhook d3d9 user32)
  add_test(NAME xy_speed_window COMMAND xy_speed_window_tests)
endif()
if(MSVC)
  foreach(target CSSXYSpeed xy_speed_loader)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
    # Public Release binaries must not embed a workstation-specific PDB path.
    target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/DEBUG:NONE>" "$<$<CONFIG:Release>:/INCREMENTAL:NO>")
  endforeach()
  if(BUILD_TESTING)
    target_compile_options(xy_speed_window_tests PRIVATE /W4 /utf-8)
  endif()
endif()
